/** Simple gradient product image with a Latin label (GD). */
function ws_demo_image( $label, $c1, $c2, $slug ) {
	if ( ! function_exists( 'imagecreatetruecolor' ) ) {
		return 0;
	}
	$w  = 600;
	$im = imagecreatetruecolor( $w, $w );
	for ( $y = 0; $y < $w; $y++ ) {
		$t = $y / $w;
		$r = (int) ( $c1[0] + ( $c2[0] - $c1[0] ) * $t );
		$g = (int) ( $c1[1] + ( $c2[1] - $c1[1] ) * $t );
		$b = (int) ( $c1[2] + ( $c2[2] - $c1[2] ) * $t );
		imageline( $im, 0, $y, $w, $y, imagecolorallocate( $im, $r, $g, $b ) );
	}
	$len = strlen( $label );
	$sm  = imagecreatetruecolor( $len * 9 + 4, 19 );
	imagealphablending( $sm, false );
	imagefill( $sm, 0, 0, imagecolorallocatealpha( $sm, 0, 0, 0, 127 ) );
	imagesavealpha( $sm, true );
	imagestring( $sm, 5, 2, 2, $label, imagecolorallocate( $sm, 255, 255, 255 ) );
	$scale = max( 2, min( 7, (int) floor( 520 / ( $len * 9 + 4 ) ) ) );
	$dw    = ( $len * 9 + 4 ) * $scale;
	$dh    = 19 * $scale;
	imagealphablending( $im, true );
	imagecopyresized( $im, $sm, (int) ( ( $w - $dw ) / 2 ), (int) ( ( $w - $dh ) / 2 ), 0, 0, $dw, $dh, $len * 9 + 4, 19 );
	$up   = wp_upload_dir();
	$file = trailingslashit( $up['path'] ) . $slug . '.png';
	imagepng( $im, $file );
	$id = wp_insert_attachment( array( 'post_mime_type' => 'image/png', 'post_title' => $label, 'post_status' => 'inherit' ), $file );
	wp_update_attachment_metadata( $id, wp_generate_attachment_metadata( $id, $file ) );
	return $id;
}
